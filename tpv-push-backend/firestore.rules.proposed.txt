rules_version = '2';
service cloud.firestore {
    match /databases/{database}/documents {

    // Función para identificar si es el Administrador principal
    function esAdmin() {
      return request.auth != null && request.auth.uid == 'e8czOfuF82VqPwNUZjMpOSMHUQo2';
    }

    // Función para identificar si el usuario autenticado es un Delivery
    function esDelivery() {
      return request.auth != null && request.auth.token.email.matches('.*delivery.*');
    }

    // Camarero: el rol NO se deduce del email, se lee de usuarios/{uid}.
    // Solo se considera camarero si el documento existe, rol == 'camarero' y activo == true.
    function esCamarero() {
      return request.auth != null
        && exists(/databases/$(database)/documents/usuarios/$(request.auth.uid))
        && get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.rol == 'camarero'
        && get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.activo == true;
    }

    // PROPOSED (S4, NOT APPLIED): narrow administrator paid-order handoff to
    // en_camino. Added by the TPV push backend proposal only. The original
    // transitions and the original esAdmin() update rule remain unchanged.
    // Requires: exact admin identity, previously unpaid order (missing pagado
    // counts as unpaid), target state en_camino, payment in efectivo/
    // transferencia/pos, cobradoEn == request.time, only payment/status
    // metadata keys changed, and sound recibido/vuelto for cash.
    // Grants nothing to any other role. Must be reviewed, applied and
    // emulator-verified manually before any client can use it.
    function esAdminCobroHandoff() {
      let prev = resource.data;
      let next = request.resource.data;
      return esAdmin()
        && prev.estado in ['nuevo', 'preparacion', 'en_camino', 'entregado']
        && next.estado == 'en_camino'
        && prev.get('pagado', false) == false
        && next.pagado == true
        && next.pago in ['efectivo', 'transferencia', 'pos']
        && next.cobradoEn == request.time
        && next.diff(prev).affectedKeys().hasOnly(['estado', 'pago', 'pagado', 'cobradoEn', 'recibido', 'vuelto'])
        && (next.pago == 'efectivo'
            ? (next.recibido is number && next.recibido >= next.total
               && next.vuelto is number && next.vuelto == next.recibido - next.total)
            : (next.recibido == null && next.vuelto == null));
    }

    // Cambios de estado permitidos en un pedido (solo hacia adelante o cancelar)
    function estadoPermitido(de, a) {
      return (de == 'nuevo'       && a in ['preparacion', 'entregado', 'cancelado'])
          || (de == 'preparacion' && a in ['en_camino', 'entregado', 'cancelado'])
          || (de == 'en_camino'   && a in ['entregado', 'cancelado'])
          || (de == 'entregado'   && a in ['cancelado', 'archivado'])
          || (de == 'cancelado'   && a == 'archivado');
    }

    // Optional web-promotion snapshot; legacy orders remain valid without it.
    // Existing rules trust client prices; this does not make them server-authoritative.
    function descuentoWebValido() {
      let d = request.resource.data;
      return (!('descuento' in d) && !('porcentajeDescuento' in d))
        || (d.keys().hasAll(['descuento', 'porcentajeDescuento'])
            && d.descuento is int && d.descuento > 0
            && d.descuento < 5000000
            && d.porcentajeDescuento in [10, 15, 20, 25]
            && d.total + d.descuento < 5000000);
    }

    // Web customers may choose pickup/delivery, never waiter-only local orders.
    function tipoWebValido() {
      let d = request.resource.data;
      return !('tipo' in d)
        || (d.tipo in ['delivery', 'retiro']
            && (d.tipo != 'retiro'
                || (d.costoDelivery is number && d.costoDelivery == 0
                    && (!('distanciaKm' in d) || d.distanciaKm == null))));
    }

    match /pedidos/{id} {
      // Cualquier cliente puede CREAR un pedido
      allow create: if request.resource.data.keys().hasOnly(
                          ['local', 'estado', 'creado', 'cliente', 'pizzas', 'extras', 'total', 'mensaje', 'pago', 'costoDelivery', 'distanciaKm', 'descuento', 'porcentajeDescuento', 'tipo'])
                    && descuentoWebValido()
                    && tipoWebValido()
                    && (!('pago' in request.resource.data) || request.resource.data.pago in ['efectivo', 'transferencia'])
                    && request.resource.data.estado == 'nuevo'
                    && request.resource.data.creado == request.time
                    && request.resource.data.total is number
                    && request.resource.data.total >= 0
                    && request.resource.data.total < 5000000
                    && request.resource.data.pizzas is list
                    && request.resource.data.pizzas.size() <= 30
                    && request.resource.data.extras is list
                    && request.resource.data.extras.size() <= 10;

      // El Administrador también puede CARGAR pedidos desde el panel (retiro, local, teléfono)
      allow create: if esAdmin()
                    && request.resource.data.estado in ['nuevo', 'preparacion', 'entregado']
                    && request.resource.data.creado == request.time
                    && request.resource.data.total is number
                    && request.resource.data.total >= 0
                    && request.resource.data.total < 5000000
                    && request.resource.data.pizzas is list
                    && request.resource.data.pizzas.size() <= 30
                    && request.resource.data.extras is list
                    && request.resource.data.extras.size() <= 10;

      // LECTURA: Administrador y Delivery como siempre.
      // El camarero SOLO puede leer pedidos locales (resource.data.tipo == 'local'),
      // por eso su consulta en el TPV debe filtrar por tipo == 'local'.
      allow read: if esAdmin() || esDelivery() || (esCamarero() && resource.data.tipo == 'local');

      // MODIFICAR: solo el administrador. El estado solo puede avanzar o cancelarse (no hay vuelta atrás).
      // 'archivado' lo usa el cierre de caja (desde entregado o cancelado).
      allow update: if esAdmin()
                    && (request.resource.data.estado == resource.data.estado
                        || estadoPermitido(resource.data.estado, request.resource.data.estado));
      // PROPOSED (S4, NOT APPLIED): administrator paid-order handoff. See esAdminCobroHandoff().
      allow update: if esAdminCobroHandoff();

      // BORRAR: solo el administrador.
      allow delete: if esAdmin();
    }

    // Nueva colección para guardar los reportes de los cierres de caja históricos
    match /cierres/{id} {
      // Solo el administrador principal puede ver o crear los cierres de caja
      allow read, write: if esAdmin();
    }

    // Gastos del día (se anotan en la caja y se archivan al cerrar)
    match /gastos/{id} {
      allow read, write: if esAdmin();
    }

    // Estado del local (abierto/cerrado, horarios y mensaje): lo lee cualquiera (la página de pedidos); solo el administrador lo escribe.
    // El menú (precios, sabores, extras) también lo lee la página de pedidos; el resto de config es solo del administrador.
    match /config/{doc} {
      // 'mesas' también es lectura pública: el TPV necesita listarlas y el camarero usarlas.
      allow read: if doc == 'local' || doc == 'menu' || doc == 'mesas' || esAdmin();
      allow write: if esAdmin();
    }

    // Memoria de clientes frecuentes (solo administrador)
    match /clientes/{id} {
      allow read, write: if esAdmin();
    }

    // Perfil de usuarios: cada usuario lee su propio documento; solo el admin lo escribe.
    match /usuarios/{uid} {
      allow read: if request.auth != null && (request.auth.uid == uid || esAdmin());
      allow write: if esAdmin();
    }

    // Cola sencilla de pedidos de mesa. El camarero propone items SIN precios;
    // el TPV administrador recalcula importes con config/menu y confirma o rechaza.
    match /solicitudes_mesa/{id} {
      // El camarero ve solo lo que él envió; el administrador ve toda la cola.
      allow read: if esAdmin() || (esCamarero() && resource.data.solicitadoPor == request.auth.uid);

      allow create: if esCamarero()
                    && request.resource.data.keys().hasOnly([
                         'local', 'estado', 'tipo', 'mesa', 'operacion', 'pedidoId',
                         'pizzas', 'extras', 'cliente', 'notas',
                         'solicitadoPor', 'solicitadoEmail', 'creado'])
                    && request.resource.data.local == 'mi-barrio'
                    && request.resource.data.estado == 'pendiente'
                    && request.resource.data.tipo == 'local'
                    && request.resource.data.operacion in ['crear', 'agregar']
                    && request.resource.data.mesa is number
                    && request.resource.data.mesa > 0
                    && exists(/databases/$(database)/documents/config/mesas)
                    && request.resource.data.mesa in get(/databases/$(database)/documents/config/mesas).data.mesas
                    && request.resource.data.pizzas is list
                    && request.resource.data.pizzas.size() <= 20
                    && request.resource.data.extras is list
                    && request.resource.data.extras.size() <= 20
                    && request.resource.data.solicitadoPor == request.auth.uid
                    && request.resource.data.creado == request.time;

      // Solo el TPV administrador confirma o rechaza (nunca el camarero).
      allow update: if esAdmin();
      allow delete: if esAdmin();
    }

    // Todo lo demás, cerrado.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
